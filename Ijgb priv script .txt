
-- Hexploit All-in-One Script (Triggerbot, Aimlock, Silent Aim, Fake Macro)
-- Compatible with Da Hood & Knockoffs (e.g., Der Hood, Dei Hood, Dee Hood)

-- Setup
local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local RunService = game:GetService("RunService")
local StarterGui = game:GetService("StarterGui")
local Camera = workspace.CurrentCamera
local LocalPlayer = Players.LocalPlayer
local Mouse = LocalPlayer:GetMouse()

-- Defaults
_G.aimlock = false
_G.silentaim = false
_G.triggerbot = false
_G.fakemacro = false

_G.aimlockKey = nil
_G.silentaimKey = nil
_G.triggerbotKey = nil
_G.fakemacroKey = nil

_G.silentAimTarget = nil
_G.aimlockTarget = nil

-- UI Setup (Using Linoria UI)
local repo = 'https://raw.githubusercontent.com/kachhow/ijgb/refs/heads/main/'
local Library = loadstring(game:HttpGet(repo .. 'UIlib.lua'))()
local ThemeManager = loadstring(game:HttpGet(repo .. 'UIthememanager.lua'))()
local SaveManager = loadstring(game:HttpGet(repo .. 'UIsavemanager.lua'))()

local Window = Library:CreateWindow({
    Title = 'Hexploit Script',
    Center = true,
    AutoShow = true,
    TabPadding = 9
})

local Tabs = {
    Main = Window:AddTab('Main')
}

local LeftGroupBox = Tabs.Main:AddLeftGroupbox('Combat')

-- UI
local features = {
    {name = "Triggerbot", var = "_G.triggerbot", key = "_G.triggerbotKey"},
    {name = "Aimlock", var = "_G.aimlock", key = "_G.aimlockKey"},
    {name = "Silent Aim", var = "_G.silentaim", key = "_G.silentaimKey"},
    {name = "Fake Macro", var = "_G.fakemacro", key = "_G.fakemacroKey"}
}

for _, feat in pairs(features) do
    LeftGroupBox:AddToggle(feat.name .. "Toggle", {
        Text = feat.name,
        Default = false,
        Tooltip = "Toggle " .. feat.name,
        Callback = function(Value)
            _G[feat.var] = Value
        end
    })

    LeftGroupBox:AddLabel(feat.name .. " Keybind"):AddKeyPicker(feat.name .. "KeyPicker", {
        Default = "",
        SyncToggleState = false,
        Mode = 'Toggle',
        Text = feat.name .. " Keybind",
        NoUI = false,
        ChangedCallback = function(New)
            _G[feat.key] = New
        end
    })
end

-- Closest player to mouse
local function getClosestPlayerToCursor()
    local closest, shortest = nil, math.huge
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local head = player.Character:FindFirstChild("Head") or player.Character:FindFirstChild("HumanoidRootPart")
            if head then
                local screenPos, onScreen = Camera:WorldToScreenPoint(head.Position)
                if onScreen then
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - UserInputService:GetMouseLocation()).Magnitude
                    if dist < shortest then
                        closest = player
                        shortest = dist
                    end
                end
            end
        end
    end
    return closest
end

-- Lock target for Silent Aim
UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == Enum.KeyCode.Q then
        local target = getClosestPlayerToCursor()
        if target then
            _G.silentAimTarget = target
            StarterGui:SetCore("SendNotification", {
                Title = "Silent Aim",
                Text = target.Name .. " officially locked on",
                Duration = 3
            })
        end
    end
end)

-- Aimlock camera follow
RunService.RenderStepped:Connect(function()
    if _G.aimlock and _G.aimlockTarget and _G.aimlockTarget.Character then
        local head = _G.aimlockTarget.Character:FindFirstChild("Head") or _G.aimlockTarget.Character:FindFirstChild("HumanoidRootPart")
        if head then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, head.Position)
        end
    end
end)

-- Silent Aim Hook
local __namecall
__namecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
    local method = getnamecallmethod()
    if method == "FireServer" and tostring(self) == "HitPart" and _G.silentaim and _G.silentAimTarget then
        local head = _G.silentAimTarget.Character and (_G.silentAimTarget.Character:FindFirstChild("Head") or _G.silentAimTarget.Character:FindFirstChild("HumanoidRootPart"))
        if head then
            local args = {...}
            args[1] = head
            return __namecall(self, unpack(args))
        end
    end
    return __namecall(self, ...)
end))

-- Triggerbot
RunService.Heartbeat:Connect(function()
    if _G.triggerbot then
        local target = Mouse.Target
        if target and target.Parent and Players:GetPlayerFromCharacter(target.Parent) then
            local tool = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildWhichIsA("Tool")
            if tool then
                tool:Activate()
            end
        end
    end
end)

-- Fake Macro
spawn(function()
    while true do
        if _G.fakemacro then
            local humanoid = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if humanoid then
                humanoid:Move(Vector3.new(1, 0, 0), false)
                task.wait(0.05)
                humanoid:Move(Vector3.new(-1, 0, 0), false)
                task.wait(0.05)
            end
        else
            task.wait(0.1)
        end
    end
end)

-- Toggle keybinds
UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.UserInputType == Enum.UserInputType.Keyboard then
        if _G.aimlockKey and input.KeyCode == _G.aimlockKey then
            _G.aimlockTarget = getClosestPlayerToCursor()
            _G.aimlock = not _G.aimlock
        end
        if _G.silentaimKey and input.KeyCode == _G.silentaimKey then
            _G.silentaim = not _G.silentaim
        end
        if _G.triggerbotKey and input.KeyCode == _G.triggerbotKey then
            _G.triggerbot = not _G.triggerbot
        end
        if _G.fakemacroKey and input.KeyCode == _G.fakemacroKey then
            _G.fakemacro = not _G.fakemacro
        end
    end
end)
